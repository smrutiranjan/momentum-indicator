 //@version=6
indicator("Mishra - Clean BUY Signals", overlay=true)

// Settings
rsiLength = input.int(14, "RSI Length")
buyLevel = input.float(60, "RSI Buy Level")
rsiMin = input.float(40, "RSI Minimum Level")
emaLength = input.int(20, "EMA Length")
maxBuy = input.int(4, "Maximum BUY Per Day", minval=1)
maxDistance = input.float(12, "Maximum Distance (Points)")

// Indicators
rsiValue = ta.rsi(close, rsiLength)
ema20 = ta.ema(close, emaLength)
vwapValue = ta.vwap(hlc3)

// Daily reset
var int buyCount = 0
var bool rsiQualified = false

if timeframe.change("1D")
    buyCount := 0
    rsiQualified := false

// CHECK 1: RSI must enter 40-60 range during same day
if rsiValue >= rsiMin and rsiValue <= buyLevel
    rsiQualified := true

// CHECK 2: RSI above 60 after same-day qualification
rsiCondition = rsiValue > buyLevel and rsiQualified

// CHECK 3: Open above both EMA20 and VWAP
priceCondition = open > ema20 and open > vwapValue

// CHECK 4: Distance to nearest indicator
emaDiff = open - ema20
vwapDiff = open - vwapValue

nearestDistance = math.min(emaDiff, vwapDiff)
distanceCondition = nearestDistance < maxDistance

// Trend classification
trendingMarket = vwapValue < ema20

// Final BUY condition
buyCondition = priceCondition and rsiCondition and distanceCondition

// Maximum BUY signals per day
buySignal = buyCondition and buyCount < maxBuy and barstate.isconfirmed

if buySignal
    buyCount += 1

// Plot EMA20 and VWAP
plot(ema20, title="EMA20", color=color.black, linewidth=1)
plot(vwapValue, title="VWAP", color=color.blue, linewidth=1)

// GREEN TRENDING BUY
plotshape(
     buySignal and trendingMarket,
     title="TRENDING BUY",
     style=shape.arrowup,
     location=location.belowbar,
     color=color.green,
     size=size.huge
)

// BLUE BUY
plotshape(
     buySignal and not trendingMarket,
     title="BUY",
     style=shape.arrowup,
     location=location.belowbar,
     color=color.blue,
     size=size.huge
)

// Alerts
alertcondition(
     buySignal and trendingMarket,
     title="TRENDING BUY Alert",
     message="TRENDING BUY - Same day RSI 40-60 qualification, RSI > 60"
)

alertcondition(
     buySignal and not trendingMarket,
     title="BUY Alert",
     message="BUY - Same day RSI 40-60 qualification, RSI > 60"
)
