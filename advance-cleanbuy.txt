 //@version=6
//Setup B
indicator("Mishra - Adv Clean BUY Signals", overlay=true)

// Settings
rsiLength = input.int(14, "RSI Length")
buyLevel = input.float(60, "RSI Buy Level")
rsiMin = input.float(40, "RSI Minimum Level")
emaLength = input.int(20, "EMA Length")
cooldownMinutes = input.int(30, "BUY Cooldown (Minutes)", minval=1)
maxBuy = input.int(4, "Maximum BUY Trades Per Day", minval=1)
maxDistance = input.float(20, "Maximum Distance (Points)")

// Indicators - Close based
rsiValue = ta.rsi(close, rsiLength)
ema20 = ta.ema(close, emaLength)
vwapValue = ta.vwap(hlc3)

// Daily reset
var bool rsiQualified = false
var int lastBuyTime = na
var int buyCount = 0

if timeframe.change("1D")
    rsiQualified := false
    lastBuyTime := na
    buyCount := 0

// CHECK 1: RSI must enter 40-60 during same day
if rsiValue >= rsiMin and rsiValue <= buyLevel
    rsiQualified := true

// CHECK 2: RSI above 60 after qualification
rsiCondition = rsiValue > buyLevel and rsiQualified

// CHECK 3: Close above EMA20 and VWAP
priceCondition = close > ema20 and close > vwapValue

// CHECK 4: Distance to nearest indicator
emaDiff = open - ema20
vwapDiff = open - vwapValue

nearestDistance = math.min(emaDiff, vwapDiff)
distanceCondition = nearestDistance < maxDistance

// Trend classification
trendingMarket = vwapValue < ema20

// CHECK 5: 30-minute cooldown
cooldownMs = cooldownMinutes * 60 * 1000

cooldownPassed = na(lastBuyTime) or
     (time - lastBuyTime >= cooldownMs)

// CHECK 6: Maximum BUY trades per day
tradeLimitPassed = buyCount < maxBuy

// FINAL BUY CONDITION
buyCondition = priceCondition and rsiCondition and distanceCondition

buySignal = buyCondition and cooldownPassed and tradeLimitPassed

// Record BUY signal
if buySignal
    lastBuyTime := time
    buyCount += 1

// Plot EMA20 and VWAP
plot(ema20, title="EMA20", color=color.black, linewidth=1)
plot(vwapValue, title="VWAP", color=color.blue, linewidth=1)

// GREEN TRENDING BUY
plotshape(
     buySignal and trendingMarket,
     title="TRENDING BUY",
     style=shape.arrowup,
     location=location.belowbar,
     color=color.green,
     size=size.huge
)

// BLUE BUY
plotshape(
     buySignal and not trendingMarket,
     title="BUY",
     style=shape.arrowup,
     location=location.belowbar,
     color=color.blue,
     size=size.huge
)

// Alerts
alertcondition(
     buySignal and trendingMarket,
     title="TRENDING BUY Alert",
     message="TRENDING BUY - All conditions satisfied"
)

alertcondition(
     buySignal and not trendingMarket,
     title="BUY Alert",
     message="BUY - All conditions satisfied"
)
